import React, { useState, useEffect, useRef } from "react";

export default function QuoteCalculator() {
  const resultRef = useRef(null);

  const [form, setForm] = useState({
    bedrooms: 3,
    bathrooms: 2,
    squareFootage: 1800,
    cleanType: "premium",
    addOns: {
      carpet: false,
      garage: false,
      blinds: false,
    },
  });

  const [price, setPrice] = useState(0);

  // 💰 PRICING ENGINE
  const calculatePrice = () => {
    let total = 120 + form.bedrooms * 25 + form.bathrooms * 20;

    if (form.squareFootage > 3000) total += 40;
    else if (form.squareFootage > 2000) total += 25;

    const multipliers = {
      standard: 1,
      premium: 1.25,
      deep: 1.5,
    };

    total *= multipliers[form.cleanType];

    const addOns = {
      carpet: 75,
      garage: 60,
      blinds: 35,
    };

    Object.keys(form.addOns).forEach((key) => {
      if (form.addOns[key]) total += addOns[key];
    });

    if (total < 150) total = 150;

    return Math.round(total / 10) * 10;
  };

  // 🔄 UPDATE PRICE
  useEffect(() => {
    setPrice(calculatePrice());
  }, [form]);

  // 🧠 AUTO-UPGRADE LOGIC
  useEffect(() => {
    const addOnCount = Object.values(form.addOns).filter(Boolean).length;

    if (form.squareFootage > 3000 || addOnCount >= 2) {
      setForm((prev) => ({ ...prev, cleanType: "deep" }));
    }
  }, [form.squareFootage, form.addOns]);

  // 🚀 AUTO SCROLL TO PRICE
  useEffect(() => {
    if (resultRef.current) {
      resultRef.current.scrollIntoView({ behavior: "smooth" });
    }
  }, [price]);

  // 🎨 CARD STYLE
  const cardStyle = (type) => {
    const active = form.cleanType === type;

    return `
      p-5 rounded-xl border cursor-pointer transition-all duration-300 text-center
      ${
        active
          ? "border-[#C6A969] shadow-xl scale-105 bg-white"
          : "border-gray-200 bg-gray-50 hover:shadow-md hover:-translate-y-1"
      }
    `;
  };

  return (
    <div className="max-w-xl mx-auto p-6">

      {/* 🏆 PACKAGE CARDS */}
      <div className="grid md:grid-cols-3 gap-4 mb-8">

        <div
          className={cardStyle("standard")}
          onClick={() => setForm({ ...form, cleanType: "standard" })}
        >
          <h3 className="font-semibold">Ongoing</h3>
          <p className="text-sm text-gray-500">Maintenance clean</p>
        </div>

        <div
          className={cardStyle("premium")}
          onClick={() => setForm({ ...form, cleanType: "premium" })}
        >
          <div className="bg-[#C6A969] text-black text-xs px-2 py-1 rounded mb-2 inline-block">
            MOST POPULAR
          </div>

          <h3 className="font-semibold text-lg">Premium</h3>
          <p className="text-sm text-gray-600">
            More detailed, polished clean
          </p>

          {form.cleanType === "premium" && (
            <ul className="text-sm mt-3 text-gray-600">
              <li>✔ Cabinet fronts</li>
              <li>✔ Baseboards (spot)</li>
              <li>✔ Mirrors & glass</li>
              <li>✔ Bed making</li>
            </ul>
          )}
        </div>

        <div
          className={cardStyle("deep")}
          onClick={() => setForm({ ...form, cleanType: "deep" })}
        >
          <h3 className="font-semibold">Deep Clean</h3>
          <p className="text-sm text-gray-500">
            Full home reset
          </p>

          {form.cleanType === "deep" && (
            <ul className="text-sm mt-3 text-gray-600">
              <li>✔ Oven & fridge</li>
              <li>✔ Grout scrubbing</li>
              <li>✔ Fixtures & fans</li>
              <li>✔ Window tracks</li>
            </ul>
          )}
        </div>

      </div>

      {/* 🏠 INPUTS */}
      <div className="grid grid-cols-3 gap-3 mb-6">

        <input
          type="number"
          value={form.bedrooms}
          onChange={(e) =>
            setForm({ ...form, bedrooms: +e.target.value })
          }
          className="border p-2 rounded"
          placeholder="Beds"
        />

        <input
          type="number"
          value={form.bathrooms}
          onChange={(e) =>
            setForm({ ...form, bathrooms: +e.target.value })
          }
          className="border p-2 rounded"
          placeholder="Baths"
        />

        <input
          type="number"
          value={form.squareFootage}
          onChange={(e) =>
            setForm({ ...form, squareFootage: +e.target.value })
          }
          className="border p-2 rounded"
          placeholder="Sq Ft"
        />

      </div>

      {/* 🧼 ADD-ONS */}
      <div className="mb-6">
        <h4 className="font-semibold mb-2">Add-Ons</h4>

        {Object.keys(form.addOns).map((key) => (
          <label key={key} className="block text-sm">
            <input
              type="checkbox"
              checked={form.addOns[key]}
              onChange={() =>
                setForm({
                  ...form,
                  addOns: {
                    ...form.addOns,
                    [key]: !form.addOns[key],
                  },
                })
              }
              className="mr-2"
            />
            {key}
          </label>
        ))}
      </div>

      {/* 💰 RESULT */}
      <div
        ref={resultRef}
        className="text-center transition-all duration-500"
      >

        <p className="text-sm text-gray-500 mb-2">
          Most homes range between $180 – $350
        </p>

        <h2 className="text-4xl font-bold text-[#1F2A37] animate-pulse">
          ${price}
        </h2>

        <p className="text-sm text-gray-500 mt-2">
          Final price confirmed after walkthrough
        </p>

        <p className="text-sm text-[#C6A969] mt-2">
          Limited availability this week — book now
        </p>

      </div>

    </div>
  );
}
